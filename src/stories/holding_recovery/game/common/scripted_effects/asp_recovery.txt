# Incident owner character; hooks call this only for an observed enemy raid/siege.
asp_holding_recovery_observe_incident_effect = {
    save_scope_as = holder
    if = { limit = { asp_recovery_origin_valid_trigger = { COUNTY = $COUNTY$ } }
        $COUNTY$ = { save_scope_as = asp_recovery_origin_county }
        if = { limit = { $COUNTY$ = { OR = { NOT = { has_variable = asp_recovery_incident_date } NOT = { var:asp_recovery_incident_date = current_date } NOT = { var:asp_recovery_incident_owner = scope:holder } } } }
            $COUNTY$ = {
                if = { limit = { NOT = { has_variable = asp_recovery_incident_serial } } set_variable = { name = asp_recovery_incident_serial value = 0 } }
                change_variable = { name = asp_recovery_incident_serial add = 1 }
                set_variable = { name = asp_recovery_incident_owner value = scope:holder }
                set_variable = { name = asp_recovery_incident_date value = current_date }
                set_variable = { name = asp_recovery_incident_control value = county_control }
                set_variable = { name = asp_recovery_incident_cause value = flag:$CAUSE$ }
            }
            asp_start_holding_recovery_effect = { COUNTY = $COUNTY$ }
        }
    }
}
asp_start_holding_recovery_effect = {
    save_scope_as = holder
    if = { limit = { asp_recovery_origin_valid_trigger = { COUNTY = $COUNTY$ } $COUNTY$ = { has_variable = asp_recovery_incident_serial var:asp_recovery_incident_owner = scope:holder OR = { NOT = { has_variable = asp_recovery_consumed_incident } NOT = { var:asp_recovery_consumed_incident = var:asp_recovery_incident_serial } } } }
        $COUNTY$ = { set_variable = { name = asp_recovery_consumed_incident value = var:asp_recovery_incident_serial } save_scope_as = asp_recovery_origin_county }
        if = { limit = { NOT = { any_owned_story = { asp_is_active_story_trigger = yes var:asp_family = flag:holding_recovery var:asp_target = $COUNTY$ } } }
            save_scope_as = asp_recovery_origin_owner
            if = { limit = { exists = scope:asp_recovery_force scope:asp_recovery_force = yes } asp_recovery_offer_or_create_effect = yes }
            else = { random = { chance = asp_recovery_selection_chance asp_recovery_offer_or_create_effect = yes } }
        }
    }
}
asp_recovery_create_effect = {
    clear_saved_scope = asp_recovery_new_story
    create_story = { type = asp_holding_recovery_story save_scope_as = asp_recovery_new_story }
    scope:asp_recovery_new_story ?= {
        asp_initialize_story_effect = { FAMILY = holding_recovery CLASS = ambition TARGET = scope:asp_recovery_origin_county }
        if = { limit = { asp_is_active_story_trigger = yes }
            set_variable = { name = asp_recovery_incident value = scope:asp_recovery_origin_county.var:asp_recovery_incident_serial }
            set_variable = { name = asp_recovery_origin_control value = scope:asp_recovery_origin_county.var:asp_recovery_incident_control }
            set_variable = { name = asp_recovery_origin_cause value = scope:asp_recovery_origin_county.var:asp_recovery_incident_cause }
            set_variable = { name = asp_interpretation value = flag:protection }
            save_scope_as = asp_recovery_event_story
            var:asp_target = {
                save_scope_as = asp_recovery_event_county
                every_county_province = { limit = { has_ongoing_construction = yes } save_scope_as = asp_recovery_old_work scope:asp_recovery_new_story = { add_to_variable_list = { name = asp_recovery_blocked_provinces target = scope:asp_recovery_old_work } } }
            }
            story_owner = { trigger_event = asp_holding_recovery.1 }
        }
    }
}
# Province callbacks retain the original payer; never substitute the current holder.
asp_holding_recovery_construction_started_effect = {
    save_scope_as = asp_recovery_work_province
    if = { limit = { NOT = { has_variable = asp_recovery_construction_serial } } set_variable = { name = asp_recovery_construction_serial value = 0 } }
    change_variable = { name = asp_recovery_construction_serial add = 1 }
    remove_variable = asp_recovery_work_story
    remove_variable = asp_recovery_work_payer
    remove_variable = asp_recovery_before_market_villages
    remove_variable = asp_recovery_allow_market_villages
    remove_variable = asp_recovery_before_palisades
    remove_variable = asp_recovery_allow_palisades
    remove_variable = asp_recovery_before_pastures
    remove_variable = asp_recovery_allow_pastures
    remove_variable = asp_recovery_before_cereal_fields
    remove_variable = asp_recovery_allow_cereal_fields
    remove_variable = asp_recovery_before_hill_farms
    remove_variable = asp_recovery_allow_hill_farms
    if = { limit = { exists = scope:character exists = county.holder }
        county.holder = { save_scope_as = holder }
        if = { limit = { scope:character = scope:holder asp_recovery_province_legal_trigger = { RESERVED = yes } }
            county.holder = {
                every_owned_story = {
                    limit = { asp_is_active_story_trigger = yes var:asp_family = flag:holding_recovery var:asp_target = scope:asp_recovery_work_province.county NOT = { has_variable = asp_recovery_lost } NOT = { has_variable = asp_recovery_completed } NOT = { any_in_list = { variable = asp_recovery_blocked_provinces this = scope:asp_recovery_work_province } } }
                    save_scope_as = asp_recovery_work_story
                    add_to_variable_list = { name = asp_recovery_started_provinces target = scope:asp_recovery_work_province }
                    scope:asp_recovery_work_province = {
                        set_variable = { name = asp_recovery_work_story value = scope:asp_recovery_work_story }
                        set_variable = { name = asp_recovery_work_payer value = scope:character }
                        set_variable = { name = asp_recovery_work_serial value = var:asp_recovery_construction_serial }
                        set_variable = { name = asp_recovery_work_started value = current_date }
                        set_variable = { name = asp_recovery_before_market_villages value = asp_recovery_market_villages_level }
                        remove_variable = asp_recovery_allow_market_villages
                        if = { limit = { asp_recovery_market_villages_legal_trigger = { RESERVED = yes } } set_variable = { name = asp_recovery_allow_market_villages value = yes } }
                        set_variable = { name = asp_recovery_before_palisades value = asp_recovery_palisades_level }
                        remove_variable = asp_recovery_allow_palisades
                        if = { limit = { asp_recovery_palisades_legal_trigger = { RESERVED = yes } } set_variable = { name = asp_recovery_allow_palisades value = yes } }
                        set_variable = { name = asp_recovery_before_pastures value = asp_recovery_pastures_level }
                        remove_variable = asp_recovery_allow_pastures
                        if = { limit = { asp_recovery_pastures_legal_trigger = { RESERVED = yes } } set_variable = { name = asp_recovery_allow_pastures value = yes } }
                        set_variable = { name = asp_recovery_before_cereal_fields value = asp_recovery_cereal_fields_level }
                        remove_variable = asp_recovery_allow_cereal_fields
                        if = { limit = { asp_recovery_cereal_fields_legal_trigger = { RESERVED = yes } } set_variable = { name = asp_recovery_allow_cereal_fields value = yes } }
                        set_variable = { name = asp_recovery_before_hill_farms value = asp_recovery_hill_farms_level }
                        remove_variable = asp_recovery_allow_hill_farms
                        if = { limit = { asp_recovery_hill_farms_legal_trigger = { RESERVED = yes } } set_variable = { name = asp_recovery_allow_hill_farms value = yes } }
                    }
                }
            }
        }
    }
}
asp_holding_recovery_construction_completed_effect = {
    save_scope_as = asp_recovery_work_province
    if = { limit = { exists = var:asp_recovery_work_story exists = scope:character has_variable = asp_recovery_work_payer var:asp_recovery_work_payer = scope:character var:asp_recovery_work_serial = var:asp_recovery_construction_serial }
        if = { limit = { var:asp_recovery_work_story = { asp_is_active_story_trigger = yes NOT = { has_variable = asp_recovery_lost } story_owner = scope:character var:asp_target = scope:asp_recovery_work_province.county } county.holder = scope:character OR = {
            AND = { has_variable = asp_recovery_allow_market_villages asp_recovery_market_villages_level > var:asp_recovery_before_market_villages }
            AND = { has_variable = asp_recovery_allow_palisades asp_recovery_palisades_level > var:asp_recovery_before_palisades }
            AND = { has_variable = asp_recovery_allow_pastures asp_recovery_pastures_level > var:asp_recovery_before_pastures }
            AND = { has_variable = asp_recovery_allow_cereal_fields asp_recovery_cereal_fields_level > var:asp_recovery_before_cereal_fields }
            AND = { has_variable = asp_recovery_allow_hill_farms asp_recovery_hill_farms_level > var:asp_recovery_before_hill_farms }
        } }
            var:asp_recovery_work_story = {
                set_variable = { name = asp_recovery_completed value = yes }
                set_variable = { name = asp_recovery_completed_province value = scope:asp_recovery_work_province }
                set_variable = { name = asp_recovery_completed_payer value = scope:character }
                set_variable = { name = asp_recovery_completed_serial value = scope:asp_recovery_work_province.var:asp_recovery_work_serial }
                set_variable = { name = asp_recovery_completed_started value = scope:asp_recovery_work_province.var:asp_recovery_work_started }
                set_variable = { name = asp_recovery_completed_date value = current_date }
            }
        }
    }
    asp_holding_recovery_construction_cancelled_effect = yes
}
asp_holding_recovery_construction_cancelled_effect = {
    save_scope_as = asp_recovery_work_province
    remove_variable = asp_recovery_work_story
    remove_variable = asp_recovery_work_payer
    county.holder ?= {
        every_owned_story = { limit = { asp_is_active_story_trigger = yes var:asp_family = flag:holding_recovery var:asp_target = scope:asp_recovery_work_province.county }
            remove_list_variable = { name = asp_recovery_blocked_provinces target = scope:asp_recovery_work_province }
        }
    }
}
# Character old holder, fixed county, exact native transfer type (unknown types are free loss).
asp_holding_recovery_title_lost_effect = {
    every_owned_story = { limit = { asp_is_active_story_trigger = yes var:asp_family = flag:holding_recovery var:asp_target = $COUNTY$ }
        set_variable = { name = asp_recovery_lost value = yes }
        set_variable = { name = asp_recovery_loss_type value = flag:$TYPE$ }
        if = { limit = { has_variable = asp_recovery_loss_type var:asp_recovery_loss_type = flag:granted } set_variable = { name = asp_recovery_granted value = yes } }
        save_scope_as = asp_recovery_losing_story
        every_in_list = { variable = asp_recovery_started_provinces
            if = { limit = { exists = var:asp_recovery_work_story var:asp_recovery_work_story = scope:asp_recovery_losing_story }
                remove_variable = asp_recovery_work_story
                remove_variable = asp_recovery_work_payer
            }
        }
    }
}
asp_holding_recovery_check_progress_effect = {
    if = { limit = { asp_is_active_story_trigger = yes has_variable = asp_recovery_incident }
        if = { limit = { OR = { has_variable = asp_recovery_lost var:asp_target = { NOT = { holder = prev.story_owner } } } }
            if = { limit = { has_variable = asp_recovery_granted } asp_resolve_story_effect = { OUTCOME = abandoned REWARDS = normal } }
            else = { asp_resolve_story_effect = { OUTCOME = impossible REWARDS = normal } }
        }
        else_if = { limit = { asp_recovery_fulfilled_trigger = yes } asp_resolve_story_effect = { OUTCOME = fulfilled REWARDS = normal } }
    }
}
asp_holding_recovery_resolve_effect = {
    if = { limit = { has_variable = asp_recovery_incident var:asp_outcome = flag:$OUTCOME$ var:asp_reward_policy = flag:$REWARDS$ }
        save_scope_as = asp_recovery_ending_story
        var:asp_target = { save_scope_as = asp_recovery_event_county }
        every_in_list = { variable = asp_recovery_started_provinces
            if = { limit = { exists = var:asp_recovery_work_story var:asp_recovery_work_story = scope:asp_recovery_ending_story }
                remove_variable = asp_recovery_work_story
                remove_variable = asp_recovery_work_payer
            }
        }
        story_owner = {
            if = { limit = { is_alive = yes }
                if = { limit = { scope:asp_recovery_ending_story = { var:asp_outcome = flag:fulfilled } }
                    create_character_memory = { type = asp_recovery_fulfilled_memory }
                    if = { limit = { scope:asp_recovery_ending_story = { var:asp_reward_policy = flag:normal } }
                        add_stewardship_lifestyle_xp = asp_recovery_stewardship_reward
                        scope:asp_recovery_event_county = { remove_county_modifier = asp_holding_recovery_tax_modifier add_county_modifier = { modifier = asp_holding_recovery_tax_modifier years = 3 } }
                    }
                }
                else_if = { limit = { scope:asp_recovery_ending_story = { var:asp_outcome = flag:impossible } } create_character_memory = { type = asp_recovery_lost_memory } }
                else = { create_character_memory = { type = asp_recovery_abandoned_memory } }
                if = { limit = { is_ai = no scope:asp_recovery_ending_story = { var:asp_reward_policy = flag:normal } }
                    save_scope_value_as = { name = asp_recovery_event_outcome value = 1 }
                    if = { limit = { scope:asp_recovery_ending_story = { var:asp_outcome = flag:impossible } } save_scope_value_as = { name = asp_recovery_event_outcome value = 2 } }
                    else_if = { limit = { scope:asp_recovery_ending_story = { NOT = { var:asp_outcome = flag:fulfilled } } } save_scope_value_as = { name = asp_recovery_event_outcome value = 3 } }
                    trigger_event = asp_holding_recovery.2
                }
            }
        }
    }
}

asp_recovery_offer_or_create_effect = {
    if = { limit = { asp_active_ambitions < asp_ambition_cap } asp_recovery_create_effect = yes }
    else = {
        save_scope_as = asp_replacement_cause_owner
        clear_saved_scope = asp_replacement_cause_victim
        clear_saved_scope = asp_replacement_cause_killer
        scope:asp_recovery_origin_county = { save_scope_as = asp_replacement_cause_county save_scope_value_as = { name = asp_replacement_cause_incident value = var:asp_recovery_incident_serial } }
        clear_saved_scope = asp_replacement_1
        clear_saved_scope = asp_replacement_2
        clear_saved_scope = asp_replacement_3
        every_owned_story = { limit = { asp_can_manage_story_trigger = yes var:asp_class = flag:ambition }
            if = { limit = { NOT = { exists = scope:asp_replacement_1 } } save_scope_as = asp_replacement_1 }
            else_if = { limit = { NOT = { exists = scope:asp_replacement_2 } } save_scope_as = asp_replacement_2 }
            else_if = { limit = { NOT = { exists = scope:asp_replacement_3 } } save_scope_as = asp_replacement_3 }
        }
        trigger_event = asp_core.2
    }
}
