asp_test_vengeance.10 = {
    type = character_event
    hidden = yes
    orphan = yes
    immediate = {
        debug_log = "ASP_PROBE VENGEANCE BEGIN"
        create_character = { name = "Probe Culprit" age = 30 gender = male random_traits = no culture = root.culture faith = root.faith employer = root save_scope_as = asp_probe_killer }
        create_character = { name = "Public Victim" age = 30 gender = male random_traits = no culture = root.culture faith = root.faith employer = root save_scope_as = asp_probe_public }
        create_character = { name = "Hidden Victim" age = 30 gender = male random_traits = no culture = root.culture faith = root.faith employer = root save_scope_as = asp_probe_hidden }
        scope:asp_probe_public = { death = { killer = scope:asp_probe_killer death_reason = death_duel } }
        scope:asp_probe_hidden = { death = { killer = scope:asp_probe_killer death_reason = death_murder } }
        if = { limit = { scope:asp_probe_public = { death_reason = death_duel killer = scope:asp_probe_killer } } debug_log = "ASP_PROBE PUBLIC PASS death_duel and exact killer" }
        else = { debug_log = "ASP_PROBE PUBLIC FAIL" }
        if = { limit = { NOT = { any_known_secret = { secret_type = secret_murder secret_owner = scope:asp_probe_killer secret_target = scope:asp_probe_hidden } } } debug_log = "ASP_PROBE HIDDEN PASS raw killer has no matching known secret" }
        else = { debug_log = "ASP_PROBE HIDDEN FAIL" }
        scope:asp_probe_killer = { add_secret = { type = secret_murder target = scope:asp_probe_hidden } random_secret = { type = secret_murder limit = { secret_target = scope:asp_probe_hidden } save_scope_as = asp_probe_secret } }
        scope:asp_probe_secret = { reveal_to = root }
        if = { limit = { any_known_secret = { secret_type = secret_murder secret_owner = scope:asp_probe_killer secret_target = scope:asp_probe_hidden is_known_by = root } } debug_log = "ASP_PROBE DISCOVERY PASS matched known secret" }
        else = { debug_log = "ASP_PROBE DISCOVERY FAIL" }
        scope:asp_probe_secret = { expose_secret = root }
        if = { limit = { any_known_secret = { secret_type = secret_murder secret_owner = scope:asp_probe_killer secret_target = scope:asp_probe_hidden is_known_by = root } } debug_log = "ASP_PROBE EXPOSURE KNOWN_SECRET_RETAINED" }
        else = { debug_log = "ASP_PROBE EXPOSURE KNOWN_SECRET_ABSENT" }
        if = { limit = { exists = scope:asp_probe_secret } debug_log = "ASP_PROBE EXPOSURE SECRET_SCOPE_RETAINED" }
        else = { debug_log = "ASP_PROBE EXPOSURE SECRET_SCOPE_ABSENT" }
        debug_log = "ASP_PROBE VENGEANCE END"
    }
}
