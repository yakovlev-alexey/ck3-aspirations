asp_test_automation_bootstrap_effect = {
    set_variable = { name = asp_test_automation_input_version value = $VERSION$ }
    # Console description evaluation precedes the setter; guard its absent variable.
    if = {
        limit = { has_variable = asp_test_automation_input_version }
        if = {
            limit = { var:asp_test_automation_input_version = 1 }
            set_variable = { name = asp_test_automation_session value = $SESSION$ }
            set_variable = { name = asp_test_automation_request value = $REQUEST$ }
            set_variable = { name = asp_test_automation_expected_request value = { value = $REQUEST$ add = 1 } }
            set_variable = { name = asp_test_automation_operation value = flag:observe }
            if = {
                limit = { NOT = { has_variable = asp_test_automation_counter } }
                set_variable = { name = asp_test_automation_counter value = 0 }
            }
            remove_variable = asp_test_automation_pending
            asp_test_automation_complete_effect = yes
        }
        else = { debug_log = "ASP_AUTOMATION|1|$SESSION$|$REQUEST$|ERROR|reason=protocol_version" }
    }
}

asp_test_automation_request_effect = {
    set_variable = { name = asp_test_automation_input_version value = $VERSION$ }
    if = {
        limit = { var:asp_test_automation_input_version = 1 var:asp_test_automation_session = $SESSION$ }
        if = {
            limit = { var:asp_test_automation_request = $REQUEST$ }
            if = {
                limit = { has_variable = asp_test_automation_pending }
                debug_log = asp_test_automation_ack
            }
            else = {
                if = { limit = { var:asp_test_automation_operation = flag:core_regression } debug_log = asp_test_automation_core_done }
                else_if = { limit = { var:asp_test_automation_operation = flag:vengeance_regression } debug_log = asp_test_automation_vengeance_done }
                else_if = { limit = { var:asp_test_automation_operation = flag:recovery_regression } debug_log = asp_test_automation_recovery_done }
                else_if = { limit = { var:asp_test_automation_operation = flag:lifecycle_regression } debug_log = asp_test_automation_lifecycle_done }
                else_if = { limit = { var:asp_test_automation_operation = flag:lifecycle_calendar_assert } debug_log = asp_test_automation_calendar_done }
                else_if = { limit = { var:asp_test_automation_operation = flag:lifecycle_reload_assert } debug_log = asp_test_automation_reload_done }
                else = { debug_log = asp_test_automation_done }
            }
        }
        else = {
            if = {
                limit = { var:asp_test_automation_expected_request = $REQUEST$ }
                set_variable = { name = asp_test_automation_request value = $REQUEST$ }
                set_variable = { name = asp_test_automation_expected_request value = { value = $REQUEST$ add = 1 } }
                debug_log = asp_test_automation_ack
                # Description evaluation still sees the previous request/operation.
                # Enter dispatch only after the request setter executes natively.
                if = { limit = { var:asp_test_automation_request = $REQUEST$ }
                set_variable = { name = asp_test_automation_operation value = flag:$OP$ }
                if = {
                    limit = { var:asp_test_automation_operation = flag:mutate }
                    change_variable = { name = asp_test_automation_counter add = 7 }
                }
                if = { limit = { var:asp_test_automation_operation = flag:core_regression } asp_test_core_regression_effect = yes }
                if = { limit = { var:asp_test_automation_operation = flag:vengeance_regression } asp_test_vengeance_regression_effect = yes }
                if = { limit = { var:asp_test_automation_operation = flag:recovery_regression } asp_test_holding_recovery_regression_effect = yes }
                if = { limit = { var:asp_test_automation_operation = flag:lifecycle_regression } asp_test_lifecycle_regression_effect = yes }
                if = { limit = { var:asp_test_automation_operation = flag:lifecycle_calendar } asp_test_lifecycle_calendar_setup_effect = yes }
                if = { limit = { var:asp_test_automation_operation = flag:lifecycle_calendar_assert } asp_test_lifecycle_calendar_assert_effect = yes }
                if = { limit = { var:asp_test_automation_operation = flag:lifecycle_reload_assert } asp_test_lifecycle_reload_assert_effect = yes }
                if = {
                    limit = { var:asp_test_automation_operation = flag:advance }
                    set_variable = asp_test_automation_pending
                    set_variable = { name = asp_test_automation_pending_session value = var:asp_test_automation_session }
                    set_variable = { name = asp_test_automation_pending_request value = var:asp_test_automation_request }
                    trigger_event = { id = asp_test_automation.2 days = $DAYS$ }
                }
                else = { asp_test_automation_complete_effect = yes }
                }
            }
            else = { debug_log = "ASP_AUTOMATION|1|$SESSION$|$REQUEST$|ERROR|reason=request_order" }
        }
    }
    else = { debug_log = "ASP_AUTOMATION|1|$SESSION$|$REQUEST$|ERROR|reason=version_or_session_identity" }
}
