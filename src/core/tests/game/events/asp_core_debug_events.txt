# Capability probe, independent of the core implementation.
asp_test_core.10 = {
    type = character_event
    title = asp_test_probe_title
    desc = asp_test_probe_desc
    theme = stewardship
    immediate = {
        debug_log = "ASP_PROBE CORE BEGIN"
        create_story = { type = asp_core_debug save_scope_as = asp_probe_story }
        scope:asp_probe_story = {
            set_variable = { name = asp_target value = root }
            set_variable = { name = asp_origin_date value = current_date }
            set_variable = { name = asp_probe_timer value = yes months = 24 }
            set_variable = { name = asp_probe_ambition_timer value = yes years = 8 }
            set_variable = { name = asp_next_review_date value = { value = current_date add = 730 } }
        }
        set_variable = { name = asp_test_probe_reference value = scope:asp_probe_story }
        set_variable = { name = asp_test_probe_origin value = current_date }
        set_variable = { name = asp_test_probe_review value = { value = current_date add = 730 } }
    }
    option = {
        name = asp_test_probe_check
        if = {
            limit = {
                scope:asp_probe_story = {
                    var:asp_target = root
                    var:asp_origin_date = root.var:asp_test_probe_origin
                    has_variable = asp_probe_timer
                    var:asp_next_review_date = root.var:asp_test_probe_review
                }
            }
            debug_log = "ASP_PROBE CORE EVENT_SCOPE PASS"
        }
        else = { debug_log = "ASP_PROBE CORE EVENT_SCOPE FAIL" }
    }
}
