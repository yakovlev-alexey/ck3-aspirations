asp_has_known_family_trigger = {
    OR = {
        var:asp_family = flag:vengeance
        var:asp_family = flag:holding_recovery
        var:asp_family = flag:personal_bond
    }
}

asp_is_active_story_trigger = {
    has_variable = asp_schema
    var:asp_schema = 1
    has_variable = asp_instance_id
    exists = var:asp_owner
    var:asp_owner = story_owner
    exists = var:asp_target
    asp_has_known_family_trigger = yes
    OR = { var:asp_class = flag:ambition var:asp_class = flag:desire }
    NOT = { has_variable = asp_resolved }
}

asp_can_manage_story_trigger = {
    asp_is_active_story_trigger = yes
    NOT = { has_variable = asp_suspended }
    story_owner = { is_alive = yes is_ai = no }
}

asp_story_review_due_trigger = {
    asp_time_until_review_days <= 0
}

# Initial age, independent of retained reminders.
asp_story_free_release_trigger = { asp_story_age_days >= asp_initial_review_days }
asp_family_can_declare_trigger = {
    OR = {
        trigger_if = { limit = { var:asp_family = flag:vengeance } asp_vengeance_can_declare_trigger = yes }
        trigger_else = { always = no }
        trigger_if = { limit = { var:asp_family = flag:holding_recovery } asp_holding_recovery_can_declare_trigger = yes }
        trigger_else = { always = no }
    }
}
asp_can_declare_story_trigger = {
    asp_can_manage_story_trigger = yes
    NOT = { has_variable = asp_declared }
    NOT = { has_variable = asp_pledge_settled }
    asp_story_age_days >= asp_declare_min_days
    asp_family_can_declare_trigger = yes
}
asp_pledge_due_trigger = {
    has_variable = asp_declared
    has_variable = asp_pledge_deadline
    asp_time_until_pledge_days <= 0
}
asp_family_public_fulfillment_trigger = {
    OR = { AND = { var:asp_family = flag:vengeance OR = { var:asp_interpretation = flag:accountability has_variable = asp_vengeance_public_fulfillment } } AND = { var:asp_family = flag:holding_recovery asp_recovery_fulfilled_trigger = yes } }
}

asp_family_has_terminal_progress_trigger = {
    OR = { AND = { var:asp_family = flag:vengeance asp_vengeance_has_terminal_progress_trigger = yes } AND = { var:asp_family = flag:holding_recovery asp_recovery_has_terminal_progress_trigger = yes } AND = { var:asp_family = flag:personal_bond asp_bond_has_terminal_progress_trigger = yes } }
}

asp_story_owned_by_event_root_trigger = {
    custom_description = { text = asp_story_owned_by_you story_owner = root }
}

# Event-character scope: never enter a weak ended Story while building a tooltip.
asp_event_can_declare_trigger = {
    trigger_if = {
        limit = { exists = scope:asp_selected_story }
        scope:asp_selected_story = { asp_can_declare_story_trigger = yes asp_story_owned_by_event_root_trigger = yes }
    }
    trigger_else = { always = no }
}

# Live event-character dispatch for already consumed major-incident replacement.
asp_replacement_cause_valid_trigger = {
    trigger_if = { limit = { exists = scope:asp_replacement_cause_county } asp_recovery_replacement_cause_valid_trigger = yes }
    trigger_else = { asp_vengeance_replacement_cause_valid_trigger = yes }
}
