        first_valid = {
            triggered_desc = { trigger = { asp_recovery_event_can_manage_trigger = { STORY = scope:asp_selected_story } scope:asp_selected_story ?= { has_variable = asp_declared } } desc = asp_recovery_declared_county_desc }

                trigger = {
                    scope:asp_selected_story ?= { asp_can_manage_story_trigger = yes var:asp_family = flag:holding_recovery }
                }
                desc = asp_recovery_private_county_desc
            }
            triggered_desc = {
                trigger = { scope:asp_selected_story ?= { asp_can_manage_story_trigger = yes has_variable = asp_declared } }
                desc = asp_management_selected_declared_character_desc
            }
            triggered_desc = {
                trigger = { scope:asp_selected_story ?= { asp_can_manage_story_trigger = yes } }
                desc = asp_management_selected_character_desc
            }
            desc = asp_management_closed_desc
        }
    }
    theme = stewardship
    option = {
        name = asp_abandon
        if = {
            limit = { scope:asp_selected_story ?= { asp_can_manage_story_trigger = yes story_owner = root } }
            if = { limit = { scope:asp_selected_story = { has_variable = asp_declared } }
                custom_tooltip = asp_public_release_cost
                show_as_tooltip = { add_prestige = asp_pledge_failure_prestige add_stress = asp_pledge_failure_stress }
            }
            else = { custom_tooltip = asp_abandon_cost }
            show_as_tooltip = {
                scope:asp_selected_story = {
                    if = { limit = { NOT = { has_variable = asp_declared } }
                    save_scope_value_as = { name = asp_preview_release_stress value = asp_abandonment_stress }
                    story_owner = { add_stress = scope:asp_preview_release_stress }
                    }
                }
            }
        }
        hidden_effect = { asp_abandon_selected_story_effect = yes }
    }
    option = {
        name = asp_declare
        trigger = { asp_event_can_declare_trigger = yes }
        custom_tooltip = asp_declare_rules
        hidden_effect = { scope:asp_selected_story ?= { if = { limit = { asp_can_declare_story_trigger = yes story_owner = root } story_owner = { trigger_event = asp_pledge.1 } } } }
    }
    option = {
        name = asp_reconsider
        trigger = { scope:asp_selected_story ?= { asp_can_manage_story_trigger = yes asp_story_owned_by_event_root_trigger = yes } }
        hidden_effect = { trigger_event = asp_management.3 }
    }
    option = {
        name = asp_withdraw
