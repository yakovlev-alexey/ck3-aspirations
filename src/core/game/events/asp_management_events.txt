asp_management.2 = {
    type = character_event
    window = big_event_window
    title = asp_management_selected_title
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = {
                    scope:asp_selected_story ?= { asp_can_manage_story_trigger = yes var:asp_family = flag:holding_recovery }
                }
                desc = asp_management_selected_county_desc
            }
            triggered_desc = {
                trigger = { scope:asp_selected_story ?= { asp_can_manage_story_trigger = yes } }
                desc = asp_management_selected_character_desc
            }
            desc = asp_management_closed_desc
        }
    }
    theme = stewardship
    option = {
        name = asp_abandon
        if = {
            limit = { scope:asp_selected_story ?= { asp_can_manage_story_trigger = yes story_owner = root } }
            custom_tooltip = asp_abandon_cost
            show_as_tooltip = {
                scope:asp_selected_story = {
                    save_scope_value_as = { name = asp_preview_release_stress value = asp_abandonment_stress }
                    story_owner = { add_stress = scope:asp_preview_release_stress }
                }
            }
        }
        hidden_effect = { asp_abandon_selected_story_effect = yes }
    }
    option = { name = asp_back hidden_effect = { asp_open_management_effect = yes } }
}
